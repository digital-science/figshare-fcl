## Overview

A specialized search input component with built-in clear and submit buttons. Supports both controlled and uncontrolled usage patterns with keyboard shortcuts and callbacks.

<Cover kind={"story"}>
  {() => (
    <div style={{ width: "100%", maxWidth: "400px" }}>
      <SearchInput 
        placeholder="Search..." 
        onChange={(e) => console.log(e.value)} 
        onSubmit={(e) => console.log("Search:", e.value)}
      />
    </div>
  )}
</Cover>

```jsx
import { SearchInput } from "@digital-science/figshare-fcl/latest/SearchInput";

export default function SearchInputOverview() {
  return (
    <SearchInput 
      placeholder="Search..." 
      onChange={(e) => console.log(e.value)} 
      onSubmit={(e) => console.log("Search:", e.value)}
    />
  );
}
```

### Controlled SearchInput

<Cover kind={"card"}>
  {() => {
    const [value, setValue] = React.useState("");
    return (
      <div style={{ display: "flex", flexDirection: "column", gap: "16px" }}>
        <h4>Controlled Component</h4>
        <div style={{ width: "100%", maxWidth: "400px" }}>
          <SearchInput 
            value={value} 
            setValue={setValue}
            placeholder="Type to search..." 
          />
        </div>

      </div>
    );
  }}
</Cover>

```jsx
import { SearchInput } from "@digital-science/figshare-fcl/latest/SearchInput";
import { useState } from "react";

export default function ControlledSearch() {
  const [value, setValue] = useState("");
  
  return (
    <div>
      <SearchInput 
        value={value} 
        setValue={setValue}
        placeholder="Type to search..." 
      />
      <p>Current value: {value}</p>
    </div>
  );
}
```

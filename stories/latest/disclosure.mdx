- [Overview](#overview)

## Overview

A flexible disclosure/accordion component using render props. Manages visibility state for expandable content with proper ARIA attributes for accessibility.

<Cover kind={"story"}>
  {() => (
    <ValueProvider initialValue={false}>
      {(isOpen, setIsOpen) => (
        <Disclosure visible={isOpen} onToggle={setIsOpen} keepMounted={false}>
          <Disclosure.Toggle>
            {(props) => (
              <Button
                {...props}
                kind="secondary"
                style={{
                  display: "flex",
                  alignItems: "center",
                  gap: "8px",
                }}
              >
                <Button.Label>Show Details</Button.Label>
                <ChevronDown
                  style={{
                    transform: props.visible ? "rotate(180deg)" : "rotate(0deg)",
                    transition: "transform 200ms ease",
                  }}
                />
              </Button>
            )}
          </Disclosure.Toggle>
          <Disclosure.Content>
            {(props) => (
              <div
                {...props}
                className={styles.content}
              >
                <p>This is the disclosed content that toggles visibility.</p>
              </div>
            )}
          </Disclosure.Content>
        </Disclosure>
      )}
    </ValueProvider>
  )}
</Cover>

The component is imported from `@digital-science/figshare-fcl/latest`.

```jsx
import { Disclosure } from "@digital-science/figshare-fcl/latest/Disclosure";
import { Button } from "@digital-science/figshare-fcl/latest/Button";
import { useState } from "react";

export default function BasicDisclosure() {
  const [isOpen, setIsOpen] = useState(false);

  return (
    <Disclosure visible={isOpen} onToggle={setIsOpen}>
      <Disclosure.Toggle>
        {(props) => (
          <Button {...props} kind="secondary">
            <Button.Label>Toggle Content</Button.Label>
          </Button>
        )}
      </Disclosure.Toggle>
      <Disclosure.Content>
        {(props) => (
          <div {...props}>
            <p>This content can be shown or hidden.</p>
          </div>
        )}
      </Disclosure.Content>
    </Disclosure>
  );
}
```

### Controlled Disclosure

<Cover kind={"card"}>
  {() => (

                <Button.Label>{isOpen ? "Hide" : "Show"} Details</Button.Label>
              </Button>
            )}
          </Disclosure.Toggle>
          <Disclosure.Content>
            {(props) => (
              <div {...props} className={styles.content}>
                <p>State is controlled by parent component.</p>
                <p>Current state: {isOpen ? "Visible" : "Hidden"}</p>
              </div>
            )}
          </Disclosure.Content>
        </Disclosure>
        <p style={{ fontSize: "12px", color: "#666" }}>
          Click "Toggle" or the button to control visibility
        </p>
      </div>
    );
  }}
  </ValueProvider>
  )}
</Cover>

```jsx
import { Disclosure } from "@digital-science/figshare-fcl/latest/Disclosure";
import { useState } from "react";

export default function ControlledDisclosure() {
  const [isOpen, setIsOpen] = useState(false);

  return (
    <div>
      <Disclosure visible={isOpen} onToggle={setIsOpen}>
        <Disclosure.Toggle>
          {(props) => (
            <button {...props}>
              {isOpen ? "Hide" : "Show"} Details
            </button>
          )}
        </Disclosure.Toggle>
        <Disclosure.Content>
          {(props) => (
            <div {...props}>
              <p>State is controlled by parent: {isOpen ? "open" : "closed"}</p>
            </div>
          )}
        </Disclosure.Content>
      </Disclosure>
    </div>
  );
}
```

```jsx
import { Disclosure } from "@digital-science/figshare-fcl/latest/Disclosure";
import { Button } from "@digital-science/figshare-fcl/latest/Button";
import { useState } from "react";

const faqs = [
  { q: "What is this?", a: "..." },
  { q: "How do I use it?", a: "..." },
  { q: "Is it accessible?", a: "..." },
];

export default function FAQDisclosure() {
  const [openIndex, setOpenIndex] = useState(null);

  return (
    <div>
      {faqs.map((faq, idx) => (
        <Disclosure
          key={idx}
          visible={openIndex === idx}
          onToggle={(isOpen) => setOpenIndex(isOpen ? idx : null)}
        >
          <Disclosure.Toggle>
            {(props) => <Button {...props}>{faq.q}</Button>}
          </Disclosure.Toggle>
          <Disclosure.Content>
            {(props) => <div {...props}>{faq.a}</div>}
          </Disclosure.Content>
        </Disclosure>
      ))}
    </div>
  );
}
```

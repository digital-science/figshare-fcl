## Overview

A flexible disclosure/accordion component using render props. Manages visibility state for expandable content with proper ARIA attributes for accessibility.

<Canvas of={DisclosureStories.Overview} />

The component is imported from `@digital-science/figshare-fcl/latest`.

## Examples

### Basic Disclosure with Button Toggle

<Canvas of={DisclosureStories.BasicDisclosureWithButtonToggle} />

```jsx
import { Disclosure } from "@digital-science/figshare-fcl/latest/Disclosure";
import { Button } from "@digital-science/figshare-fcl/latest/Button";
import { useState } from "react";

export default function BasicDisclosure() {
  const [isOpen, setIsOpen] = useState(false);

  return (
    <Disclosure visible={isOpen} onToggle={setIsOpen}>
      <Disclosure.Toggle>
        {(props) => (
          <Button {...props} kind="secondary">
            <Button.Label>Toggle Content</Button.Label>
          </Button>
        )}
      </Disclosure.Toggle>
      <Disclosure.Content>
        {(props) => (
          <div {...props}>
            <p>This content can be shown or hidden.</p>
          </div>
        )}
      </Disclosure.Content>
    </Disclosure>
  );
}
```

### With Icon Rotation

<Canvas of={DisclosureStories.WithIconRotation} />

```jsx
import { Disclosure } from "@digital-science/figshare-fcl/latest/Disclosure";
import { Button } from "@digital-science/figshare-fcl/latest/Button";
import { ChevronDown } from "@digital-science/figshare-fcl/icons/react";
import { useState } from "react";

export default function DisclosureWithIcon() {
  const [isOpen, setIsOpen] = useState(false);

  return (
    <Disclosure visible={isOpen} onToggle={setIsOpen}>
      <Disclosure.Toggle>
        {(props) => (
          <Button
            {...props}
            kind="tertiary"
            style={{
              display: "flex",
              alignItems: "center",
              gap: "8px",
            }}
          >
            <ChevronDown
              style={{
                transform: props.visible ? "rotate(180deg)" : "rotate(0deg)",
                transition: "transform 200ms ease",
              }}
            />
            <Button.Label>Advanced Options</Button.Label>
          </Button>
        )}
      </Disclosure.Toggle>
      <Disclosure.Content>
        {(props) => (
          <div {...props}>
            <p>Additional options go here.</p>
          </div>
        )}
      </Disclosure.Content>
    </Disclosure>
  );
}
```

### Controlled Disclosure

<Canvas of={DisclosureStories.ControlledDisclosure} />

```jsx
import { Disclosure } from "@digital-science/figshare-fcl/latest/Disclosure";
import { useState } from "react";

export default function ControlledDisclosure() {
  const [isOpen, setIsOpen] = useState(false);

  return (
    <div>
      <Disclosure visible={isOpen} onToggle={setIsOpen}>
        <Disclosure.Toggle>
          {(props) => (
            <button {...props}>
              {isOpen ? "Hide" : "Show"} Details
            </button>
          )}
        </Disclosure.Toggle>
        <Disclosure.Content>
          {(props) => (
            <div {...props}>
              <p>State is controlled by parent: {isOpen ? "open" : "closed"}</p>
            </div>
          )}
        </Disclosure.Content>
      </Disclosure>
    </div>
  );
}
```

### Multiple Disclosures (FAQ Pattern)

<Canvas of={DisclosureStories.MultipleDisclosuresFAQPattern} />


```jsx
import { Disclosure } from "@digital-science/figshare-fcl/latest/Disclosure";
import { Button } from "@digital-science/figshare-fcl/latest/Button";
import { useState } from "react";

const faqs = [
  { q: "What is this?", a: "..." },
  { q: "How do I use it?", a: "..." },
  { q: "Is it accessible?", a: "..." },
];

export default function FAQDisclosure() {
  const [openIndex, setOpenIndex] = useState(null);

  return (
    <div>
      {faqs.map((faq, idx) => (
        <Disclosure
          key={idx}
          visible={openIndex === idx}
          onToggle={(isOpen) => setOpenIndex(isOpen ? idx : null)}
        >
          <Disclosure.Toggle>
            {(props) => <Button {...props}>{faq.q}</Button>}
          </Disclosure.Toggle>
          <Disclosure.Content>
            {(props) => <div {...props}>{faq.a}</div>}
          </Disclosure.Content>
        </Disclosure>
      ))}
    </div>
  );
}
```

## Overview

A flexible icon wrapper component for rendering icons with customizable blending and sizing. Supports passing icons as components or directly rendering icon elements.

<Cover kind={"story"}>
  {() => (
    <div style={{ display: "flex", gap: "24px", padding: "24px" }}>
      <IconSlot><Eye /></IconSlot>
      <IconSlot><Checkmark /></IconSlot>
      <IconSlot><Delete /></IconSlot>
    </div>
  )}
</Cover>

```jsx
import { IconSlot } from "@digital-science/figshare-fcl/latest/IconSlot";
import { Eye, Checkmark, Delete } from "@digital-science/figshare-fcl/icons/react";

export default function IconSlotOverview() {
  return (
    <div style={{ display: "flex", gap: "24px" }}>
      <IconSlot><Eye /></IconSlot>
      <IconSlot><Checkmark /></IconSlot>
      <IconSlot><Delete /></IconSlot>
    </div>
  );
}
```

The component can be imported from `@digital-science/figshare-fcl/latest`.
